 String ticarne;
        double valor=0,total=0,peso=0;
        
        JOptionPane.showMessageDialog(null,"Bienvenidos a su Carniceria!! ");
        
        ticarne=JOptionPane.showInputDialog("Digite el tipo de Carne : "+ "\n" +
                                                        "Cerdo"+ "\n" +
                                                        "Res" + "\n" +
                                                        "Bufalo");
        
        
        switch (ticarne.toLowerCase()){
   
            case "cerdo": 
                peso=Double.parseDouble(JOptionPane.showInputDialog("Digite el peso de su compra"));
                    if (peso>=5){
                     valor=24000;
                    }else if (peso<5){
                     valor=28000;
                    }
               break;
            case "res":
                peso=Double.parseDouble(JOptionPane.showInputDialog("Digite el peso de su compra"));
                    if (peso>=10){
                     valor=30000;
                    }else if (peso<10){
                     valor=35000;
                    }
                 break;    
            case "bufalo":
                peso=Double.parseDouble(JOptionPane.showInputDialog("Digite el peso de su compra"));
                 
                    if (peso>=7){
                     valor=40000;
                    }else if (peso<7){
                     valor=43000;
                    }
                    break;
                    
            default:
                JOptionPane.showMessageDialog(null,"Error!! ");
                
        
        }
        
        total=valor*peso;
        
        JOptionPane.showMessageDialog(null,"El tipo de carne Comprado es : "+ ticarne+"\n"+
        "La cantidad compradad en kilos es  : "+ peso+"\n"+
        "El toral de su compra es : "+ total);
